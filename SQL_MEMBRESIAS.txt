-- ============================================
-- Sistema de Gestión de Membresías
-- ============================================

-- Tabla de membresías de usuarios
CREATE TABLE IF NOT EXISTS user_memberships (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID NOT NULL,
  status TEXT NOT NULL DEFAULT 'active', -- active, grace_period, expired, suspended
  starts_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  expires_at TIMESTAMPTZ NOT NULL,
  grace_period_until TIMESTAMPTZ, -- Fecha límite de prórroga
  grace_period_notes TEXT, -- Notas del acuerdo de prórroga
  monthly_fee DECIMAL(10,2) DEFAULT 5000.00, -- Precio mensual acordado
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Tabla de pagos registrados
CREATE TABLE IF NOT EXISTS membership_payments (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID NOT NULL,
  membership_id UUID REFERENCES user_memberships(id),
  amount DECIMAL(10,2) NOT NULL,
  payment_date TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  payment_method TEXT, -- transferencia, efectivo, mercadopago, etc.
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Índices para queries rápidas
CREATE INDEX IF NOT EXISTS idx_user_memberships_user_id ON user_memberships(user_id);
CREATE INDEX IF NOT EXISTS idx_user_memberships_status ON user_memberships(status);
CREATE INDEX IF NOT EXISTS idx_user_memberships_expires_at ON user_memberships(expires_at);
CREATE INDEX IF NOT EXISTS idx_membership_payments_user_id ON membership_payments(user_id);
CREATE INDEX IF NOT EXISTS idx_membership_payments_payment_date ON membership_payments(payment_date);

-- Habilitar RLS
ALTER TABLE user_memberships ENABLE ROW LEVEL SECURITY;
ALTER TABLE membership_payments ENABLE ROW LEVEL SECURITY;

-- Políticas de seguridad (solo admins pueden escribir, usuarios pueden leer lo suyo)
CREATE POLICY "Users can view their own membership"
  ON user_memberships FOR SELECT
  USING (auth.uid() = user_id);

CREATE POLICY "Admins can manage all memberships"
  ON user_memberships FOR ALL
  USING (true)
  WITH CHECK (true);

CREATE POLICY "Users can view their own payments"
  ON membership_payments FOR SELECT
  USING (auth.uid() = user_id);

CREATE POLICY "Admins can manage all payments"
  ON membership_payments FOR ALL
  USING (true)
  WITH CHECK (true);

-- Trigger para actualizar updated_at
CREATE OR REPLACE FUNCTION update_membership_updated_at()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS update_user_memberships_updated_at ON user_memberships;
CREATE TRIGGER update_user_memberships_updated_at
  BEFORE UPDATE ON user_memberships
  FOR EACH ROW
  EXECUTE FUNCTION update_membership_updated_at();

-- Función para obtener estadísticas de ganancias
CREATE OR REPLACE FUNCTION get_revenue_stats()
RETURNS TABLE (
  total_revenue DECIMAL,
  monthly_revenue DECIMAL,
  active_users INTEGER,
  grace_period_users INTEGER,
  expired_users INTEGER
) AS $$
BEGIN
  RETURN QUERY
  SELECT 
    COALESCE(SUM(mp.amount), 0) as total_revenue,
    COALESCE(SUM(CASE WHEN mp.payment_date >= DATE_TRUNC('month', CURRENT_DATE) THEN mp.amount ELSE 0 END), 0) as monthly_revenue,
    COUNT(CASE WHEN um.status = 'active' THEN 1 END)::INTEGER as active_users,
    COUNT(CASE WHEN um.status = 'grace_period' THEN 1 END)::INTEGER as grace_period_users,
    COUNT(CASE WHEN um.status = 'expired' THEN 1 END)::INTEGER as expired_users
  FROM user_memberships um
  LEFT JOIN membership_payments mp ON mp.user_id = um.user_id;
END;
$$ LANGUAGE plpgsql;

-- Comentarios
COMMENT ON TABLE user_memberships IS 'Membresías de usuarios con gestión de prórrogas';
COMMENT ON TABLE membership_payments IS 'Pagos registrados de membresías';
COMMENT ON COLUMN user_memberships.grace_period_until IS 'Fecha límite de prórroga acordada con el usuario';
COMMENT ON COLUMN user_memberships.grace_period_notes IS 'Notas del acuerdo de prórroga (ej: "Acordó pagar el 15/09")';
